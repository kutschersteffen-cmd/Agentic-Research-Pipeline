================================================================================
 LOGIC OF TARGETS (Q2.1) - PSEUDO CODE
 Source: framework.json (what Decision Studio runs). One run per company.
================================================================================

CONVENTIONS
    answer values      : "Yes" | "No" | "No info"
    "has info"         : the normalised value is "Yes" or "No" (not "No info")
    source priority    : SBTi > MSCI > TPI / CA100+ > WBA > CDP
    a source's "No"    : ends the search (it is an answer, not a gap)
    all sources blank  : "No info"


--------------------------------------------------------------------------------
 STEP 1  NORMALISE   (Rules tab, expression node)
--------------------------------------------------------------------------------

FUNCTION yes_no(x):
    IF x is true  OR lower(x) == "yes"  THEN RETURN "Yes"
    IF x is false OR lower(x) == "no"   THEN RETURN "No"
    RETURN "No info"

FOR EACH input column c IN
        ca100_3_1, ca100_3_2a,
        cdp_q2_1_1 .. cdp_q2_1_4,
        msci_q2_1_1 .. msci_q2_1_4,
        tpi_q4l2,
        wba_targets, wba_s12_near_term_aim, wba_s12_covers_95,
        wba_s3_material_categories:
    <c>_answer = yes_no(c)

sbti_answer =
    IF lower(sbti_near_term_status) == "targets set"
       AND 2029 <= sbti_near_term_target_year <= 2035
    THEN "Yes"
    ELSE "No info"                       # SBTi never answers "No"

ca100_3_1_or_wba_s12_near_term_aim_answer =
    IF   yes_no(ca100_3_1) == "Yes" OR yes_no(wba_s12_near_term_aim) == "Yes"  THEN "Yes"
    ELIF yes_no(ca100_3_1) == "No"  OR yes_no(wba_s12_near_term_aim) == "No"   THEN "No"
    ELSE "No info"

tpi_alignment_answer =
    IF   alignment_2030 IN ("1.5 degrees", "below 2 degrees")
      OR alignment_2035 IN ("1.5 degrees", "below 2 degrees")     THEN "Yes"
    ELIF alignment_2030 is blank AND alignment_2035 is blank      THEN "No info"
    ELSE "No"


--------------------------------------------------------------------------------
 STEP 2  Q2.1.1 - Q2.1.4   (Rules tab, decision tables, hit policy: FIRST)
--------------------------------------------------------------------------------

Same shape for each question; only the source columns differ.

Q2.1.1  Quantitative GHG reduction targets set?
    IF   sbti_answer == "Yes"            THEN q2_1_1 = "Yes",                    source = "SBTi"
    ELIF msci_q2_1_1_answer has info     THEN q2_1_1 = msci_q2_1_1_answer,       source = "MSCI"
    ELIF tpi_q4l2_answer has info        THEN q2_1_1 = tpi_q4l2_answer,          source = "TPI"
    ELIF wba_targets_answer has info     THEN q2_1_1 = wba_targets_answer,       source = "WBA"
    ELIF cdp_q2_1_1_answer has info      THEN q2_1_1 = cdp_q2_1_1_answer,        source = "CDP"
    ELSE                                      q2_1_1 = "No info",                source = "none"

Q2.1.2  At least one interim target for 2029-2035?
    IF   sbti_answer == "Yes"            THEN q2_1_2 = "Yes",                    source = "SBTi"
    ELIF msci_q2_1_2_answer has info     THEN q2_1_2 = msci_q2_1_2_answer,       source = "MSCI"
    ELIF ca100_3_1_answer has info       THEN q2_1_2 = ca100_3_1_answer,         source = "CA100+"
    ELIF wba_s12_near_term_aim_answer has info
                                         THEN q2_1_2 = wba_s12_near_term_aim_answer, source = "WBA"
    ELIF cdp_q2_1_2_answer has info      THEN q2_1_2 = cdp_q2_1_2_answer,        source = "CDP"
    ELSE                                      q2_1_2 = "No info",                source = "none"

Q2.1.3  Interim targets cover >= 95% of Scope 1 and 2?
    IF   sbti_answer == "Yes"            THEN q2_1_3 = "Yes",                    source = "SBTi"
    ELIF msci_q2_1_3_answer has info     THEN q2_1_3 = msci_q2_1_3_answer,       source = "MSCI"
    ELIF ca100_3_2a_answer has info      THEN q2_1_3 = ca100_3_2a_answer,        source = "CA100+"
    ELIF wba_s12_covers_95_answer has info
                                         THEN q2_1_3 = wba_s12_covers_95_answer, source = "WBA"
    ELIF cdp_q2_1_3_answer has info      THEN q2_1_3 = cdp_q2_1_3_answer,        source = "CDP"
    ELSE                                      q2_1_3 = "No info",                source = "none"

Q2.1.4  Scope 3 targets, where appropriate?
    IF   sbti_answer == "Yes"            THEN q2_1_4 = "Yes",                    source = "SBTi"
    ELIF msci_q2_1_4_answer has info     THEN q2_1_4 = msci_q2_1_4_answer,       source = "MSCI"
    ELIF ca100_3_1_or_wba_s12_near_term_aim_answer has info
                                         THEN q2_1_4 = ca100_3_1_or_wba_s12_near_term_aim_answer,
                                              source = "CA100+/WBA"
    ELIF wba_s3_material_categories_answer has info
                                         THEN q2_1_4 = wba_s3_material_categories_answer,
                                              source = "WBA"
    ELIF cdp_q2_1_4_answer has info      THEN q2_1_4 = cdp_q2_1_4_answer,        source = "CDP"
    ELSE                                      q2_1_4 = "No info",                source = "none"


--------------------------------------------------------------------------------
 STEP 3  Q2.1.5   (decision table, hit policy: FIRST)
         Validated or aligned with 1.5C / well-below-2C?
--------------------------------------------------------------------------------

    IF   sbti_answer == "Yes"            THEN q2_1_5 = "Yes",     source = "SBTi"
    ELIF any of q2_1_1..q2_1_4 == "No"   THEN q2_1_5 = "No",      source = "Q2.1.1-4"
    ELIF any of q2_1_1..q2_1_4 == "No info"
                                         THEN q2_1_5 = "No info", source = "Q2.1.1-4"
    ELIF tpi_alignment_answer has info   THEN q2_1_5 = tpi_alignment_answer, source = "TPI"
    ELSE                                      q2_1_5 = "No info", source = "none"

    # the checks run in table order: Q2.1.1 No, 2 No, 3 No, 4 No, then the same for "No info"


--------------------------------------------------------------------------------
 STEP 4  COUNTS   (Rules tab, expression node)
--------------------------------------------------------------------------------

    answers       = [q2_1_1, q2_1_2, q2_1_3, q2_1_4, q2_1_5]
    yes_count     = COUNT(answers == "Yes")
    no_info_count = COUNT(answers == "No info")
    note          = yes_count + "/5 Yes"
                    + FOR EACH question qn:  " · Qn.n <answer> (<source>)"


--------------------------------------------------------------------------------
 STEP 5  OUTCOME   (Rules tab, decision table, hit policy: FIRST)
--------------------------------------------------------------------------------

    IF   yes_count == 5                                        THEN outcome = "aligned"
    ELIF q2_1_1 == q2_1_2 == q2_1_3 == q2_1_4 == "Yes"         THEN outcome = "not_aligned"
    ELIF yes_count > 0                                         THEN outcome = "partial"
    ELIF no_info_count == 5                                    THEN outcome = "no_info"
    ELSE                                                            outcome = "none"


--------------------------------------------------------------------------------
 STEP 6  TIER   (Decision tree tab, decision table)
--------------------------------------------------------------------------------

    outcome        tier   name
    -------------  -----  -----------------------------
    aligned          1    Aligned targets
    not_aligned      2    Targets, alignment not shown
    partial          3    Partial targets
    none             4    No targets
    no_info          5    No information

    RETURN tier, note
